find_package(GTest QUIET)
if(NOT GTest_FOUND)
    include(FetchContent)
    FetchContent_Declare(googletest
            URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
    add_library(GTest::GTest ALIAS gtest)
    add_library(GTest::Main ALIAS gtest_main)
endif()

macro(make_units_test name)
    add_executable(${name} utl/${name}.cpp)
    target_link_libraries(${name} utl::utl GTest::GTest GTest::Main)
    utl_apply_warnings(${name})
    add_test(NAME ${name} COMMAND ${name})
endmacro()

function(make_compile_fail_test name expected_message)
    add_executable(${name} EXCLUDE_FROM_ALL compile_fail/${name}.cpp)
    target_link_libraries(${name} utl::utl)
    add_test(NAME ${name}
             COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target ${name} --config $<CONFIG>)
    set_tests_properties(${name} PROPERTIES PASS_REGULAR_EXPRESSION "${expected_message}" RUN_SERIAL TRUE)
endfunction()
